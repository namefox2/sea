    └── AtmosphericHazeVolume        (Volume — Local/Global, Fog override)
            → HorizonHazeRenderer.cs

    ├── OceanRenderer                (Crest.OceanRenderer)
    │       → RealisticOceanSetup.cs
    │       → OceanDynamicsController.cs
    ├── ShapeFFT_WindChop            (Crest.ShapeFFT)  WindSpeed=10  JONSWAP
    ├── ShapeFFT_Swell               (Crest.ShapeFFT)  WindSpeed=12  TMA  Dir=220°
    ├── OceanDepthCache              (Crest.OceanDepthCache)
    └── RegisterDynWaves             (Crest.RegisterDynWavesInput)

▼ [VFX]
    ├── SprayParticles               (ParticleSystem — sea spray at crests)
    │       → SeaSprayParticleDriver.cs
    ├── MistParticles                (ParticleSystem — fine mist layer)
    ├── FoamLineParticles            (ParticleSystem — long foam streaks)
    └── CausticsProjector           (Projector)
            → CausticsProjector.cs
            Parent: DirectionalLight_Sun (follows sun angle automatically)

▼ [Camera]
    ├── CinematicCamera              (Camera, HDAdditionalCameraData, AudioListener)
    │       → CinematicOceanCamera.cs
    │       → UnderwaterRenderer.cs
    │       → AudioLowPassFilter (cutoff=22000, resonance=1)
    └── PostProcessVolume            (Volume — Local, follows camera)
            → OceanPostProcessingController.cs
            Overrides: Bloom, Tonemapping(ACES), ColorAdjustments,
                       ChromaticAberration, Vignette, MotionBlur,
                       LensDistortion, DepthOfField, SSR

▼ [Weather]
    └── WeatherSystem                (empty GameObject)
            → OceanWeatherSystem.cs (ref: Sun, Volume, OceanSetup)
            → OceanAudioManager.cs

▼ [Audio]
    ├── AudioSource_DeepSwell        (AudioSource, 3D=off, loop, clip=ocean_swell.wav)
    ├── AudioSource_ChopWaves        (AudioSource, 3D=off, loop, clip=ocean_chop.wav)
    ├── AudioSource_FoamSurf         (AudioSource, 3D=off, loop, clip=ocean_foam.wav)
    ├── AudioSource_WindHiss         (AudioSource, 3D=off, loop, clip=wind_hiss.wav)
    ├── AudioSource_Seagulls         (AudioSource, 3D=on,  loop, clip=seagulls.wav)
    ├── AudioSource_Rain             (AudioSource, 3D=off, loop, clip=rain_heavy.wav)
    └── AudioSource_Underwater       (AudioSource, 3D=off, loop, clip=underwater_amb.wav)
